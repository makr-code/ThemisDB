1. Get the application URL by running these commands:
{{- if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  {{- range .paths }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}
{{- end }}
{{- else if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "themisdb.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
     NOTE: It may take a few minutes for the LoadBalancer IP to be available.
           You can watch the status of by running 'kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "themisdb.fullname" . }}'
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "themisdb.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.service.port }}
{{- else if contains "ClusterIP" .Values.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "themisdb.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  echo "Visit http://127.0.0.1:8080 to use your application"
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8080:8080
{{- end }}

2. ThemisDB is now running. Useful endpoints:

   HTTP API:     {{ if .Values.ingress.enabled }}https://{{ (first .Values.ingress.hosts).host }}{{ else }}http://localhost:8080{{ end }}
   Health check: http://localhost:{{ .Values.themisdb.healthPort | default 8081 }}/health
   Metrics:      http://localhost:{{ .Values.service.metricsPort | default 9091 }}/metrics

3. Prometheus / ServiceMonitor
{{- if .Values.metrics.serviceMonitor.enabled }}
   ServiceMonitor is ENABLED — Prometheus Operator will scrape the 'metrics' port ({{ .Values.service.metricsPort | default 9091 }}).
   Namespace: {{ .Values.metrics.serviceMonitor.namespace | default .Release.Namespace }}
   Interval:  {{ .Values.metrics.serviceMonitor.interval | default "30s" }}
{{- else }}
   ServiceMonitor is DISABLED. To enable Prometheus Operator scraping:

     helm upgrade {{ .Release.Name }} themisdb/themisdb \
       --set metrics.serviceMonitor.enabled=true \
       --set metrics.serviceMonitor.namespace=monitoring

   Or add the following to your values file and upgrade:
     metrics:
       serviceMonitor:
         enabled: true
         namespace: monitoring
{{- end }}

4. Grafana Dashboards
{{- if .Values.grafana.dashboards.enabled }}
   Dashboard ConfigMaps have been created with label {{ .Values.grafana.dashboards.label | default "grafana_dashboard" }}={{ .Values.grafana.dashboards.labelValue | default "1" }}.
   Make sure your Grafana deployment has sidecar.dashboards.enabled=true to auto-import them.
   Namespace: {{ .Values.grafana.dashboards.namespace | default .Release.Namespace }}
{{- else }}
   Grafana dashboard ConfigMaps are DISABLED. To enable:

     helm upgrade {{ .Release.Name }} themisdb/themisdb \
       --set grafana.dashboards.enabled=true

   Requires Grafana sidecar (grafana.sidecar.dashboards.enabled=true in the Grafana chart).
{{- end }}
