cmake_minimum_required(VERSION 3.20)
project(user_storage_encrypted_tests)

find_package(GTest QUIET)
if(NOT GTest_FOUND)
    message(WARNING "GTest not found; user_storage_encrypted tests skipped.")
    return()
endif()

file(GLOB TEST_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/test_*.cpp")
foreach(_src IN LISTS TEST_SOURCES)
    get_filename_component(_stem "${_src}" NAME_WE)
    add_executable(use_${_stem} ${_src})
    target_link_libraries(use_${_stem} PRIVATE user_storage_encrypted GTest::gtest_main)
    add_test(NAME use_${_stem} COMMAND use_${_stem})
    set_tests_properties(use_${_stem} PROPERTIES TIMEOUT 120 LABELS "user_storage_encrypted")
endforeach()
