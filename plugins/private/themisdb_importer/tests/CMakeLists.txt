cmake_minimum_required(VERSION 3.20)
project(themisdb_importer_tests)

find_package(GTest QUIET)
if(NOT GTest_FOUND)
    message(WARNING "GTest not found; importer tests skipped.")
    return()
endif()

# All importer shared + individual importer libs must be available
set(_IMPORTER_LIBS themisdb_importer_shared)
foreach(_mod mysql mongo kafka s3)
    if(TARGET ${_mod}_importer)
        list(APPEND _IMPORTER_LIBS ${_mod}_importer)
    endif()
endforeach()

file(GLOB TEST_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/test_*.cpp")
foreach(_src IN LISTS TEST_SOURCES)
    get_filename_component(_stem "${_src}" NAME_WE)
    set(_target "importer_${_stem}")
    add_executable(${_target} ${_src})
    target_link_libraries(${_target} PRIVATE ${_IMPORTER_LIBS} GTest::gtest_main)
    add_test(NAME ${_target} COMMAND ${_target})
    set_tests_properties(${_target} PROPERTIES TIMEOUT 120 LABELS "importer")
endforeach()
