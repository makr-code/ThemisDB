   61: static EpochFencingConfig makeConfig(const std::string& shard = "s1",
   62:                                      const std::string& node  = "n1",
   63:                                      bool auto_stonith         = true) {
   64:     EpochFencingConfig cfg;
   65:     cfg.shard_id     = shard;
   66:     cfg.node_id      = node;
   67:     cfg.auto_stonith = auto_stonith;
   68:     cfg.stonith_timeout_ms = 500ms;
   69:     return cfg;
   70: }

   73:                                    std::chrono::milliseconds renew   = 50ms,
   74:                                    std::chrono::milliseconds wait    = 500ms) {
   75:     LeaseConfig cfg;
   76:     cfg.ttl_ms          = ttl;
   77:     cfg.renew_before_ms = renew;
   78:     cfg.acquire_wait_ms = wait;
   79:     return cfg;
   80: }
