# Erstelle eine Test-Konfiguration die nur MINIMALE Module einbindet
cmake_minimum_required(VERSION 3.20)
project(themis_server_debug_minimal)

# Minimale Konfiguration - nur Basis-Abhängigkeiten
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Nur die kritischsten Quellen
file(GLOB_RECURSE MINIMAL_SOURCES
    "src/main_server.cpp"
    "src/utils/logger.cpp"
    "src/utils/logger_impl.cpp"
)

add_executable(themis_minimal ${MINIMAL_SOURCES})

# Nur Basis-Includes
target_include_directories(themis_minimal PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/include"
)

# Minimale Compiler-Optionen
if(MSVC)
    target_compile_options(themis_minimal PRIVATE
        /W4
        /EHsc
        /DEBUG:FULL
    )
else()
    target_compile_options(themis_minimal PRIVATE
        -Wall -Wextra -g3
    )
endif()

message(STATUS "Built minimal themis_server for debugging")
