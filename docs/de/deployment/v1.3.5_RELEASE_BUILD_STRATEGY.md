## 🔨 Build Commands v1.3.5

### 1. Community Edition (Production)

```powershell
# Windows
cmake -B build-community -G "Visual Studio 17 2022" -A x64 `
  -DCMAKE_TOOLCHAIN_FILE="vcpkg/scripts/buildsystems/vcpkg.cmake" `
  -DTHEMIS_EDITION="COMMUNITY" `
  -DTHEMIS_ENABLE_GPU=ON `
  -DTHEMIS_ENABLE_LLM=ON `
  -DCMAKE_BUILD_TYPE=Release
cmake --build build-community --config Release --parallel 8

# Linux
cmake -B build-community -DTHEMIS_EDITION="COMMUNITY" `
  -DTHEMIS_ENABLE_GPU=ON `
  -DTHEMIS_ENABLE_LLM=ON `
  -DCMAKE_BUILD_TYPE=Release
cmake --build build-community -j$(nproc)

# Docker (multi-arch)
docker buildx build -f Dockerfile.community \
  --platform linux/amd64,linux/arm64 \
  -t themisdb:1.3.5-community \
  -t themisdb:latest \
  --push .
```

## 🏗️ Unified Build System (Single CMakeLists.txt)

### CMake Configuration Approach

```cmake
# CMakeLists.txt (unified for all editions)

# Edition selection (required)
set(THEMIS_EDITION "COMMUNITY" CACHE STRING "Edition: COMMUNITY, ENTERPRISE, HYPERSCALER")

# Feature flags (automatically set based on edition)
if(THEMIS_EDITION STREQUAL "COMMUNITY")
    set(THEMIS_ENABLE_SHARDING OFF)
    set(THEMIS_ENABLE_MULTI_MASTER OFF)
    set(THEMIS_ENABLE_RBAC OFF)
    set(THEMIS_ENABLE_HSM OFF)
    set(THEMIS_ENABLE_AUTO_FAILOVER OFF)
    set(THEMIS_ENABLE_COMPLIANCE_AUDIT OFF)
    
    # Community gets GPU & LLM
    set(THEMIS_ENABLE_GPU ON)
    set(THEMIS_ENABLE_LLM ON)
    
    message(STATUS "Building ThemisDB COMMUNITY Edition (80% coverage)")
    target_compile_definitions(themis_core PUBLIC THEMIS_EDITION_COMMUNITY)

elseif(THEMIS_EDITION STREQUAL "ENTERPRISE")
    set(THEMIS_ENABLE_SHARDING ON)
    set(THEMIS_ENABLE_MULTI_MASTER ON)
    set(THEMIS_ENABLE_RBAC ON)
    set(THEMIS_ENABLE_HSM ON)
    set(THEMIS_ENABLE_AUTO_FAILOVER ON)
    set(THEMIS_ENABLE_COMPLIANCE_AUDIT ON)
    
    # Enterprise inherits all Community features
    set(THEMIS_ENABLE_GPU ON)
    set(THEMIS_ENABLE_LLM ON)
    
    message(STATUS "Building ThemisDB ENTERPRISE Edition")
    target_compile_definitions(themis_core PUBLIC THEMIS_EDITION_ENTERPRISE)
    add_subdirectory(src/enterprise)

elseif(THEMIS_EDITION STREQUAL "HYPERSCALER")
    # All flags ON
    set(THEMIS_ENABLE_SHARDING ON)
    set(THEMIS_ENABLE_MULTI_MASTER ON)
    set(THEMIS_ENABLE_RBAC ON)
    set(THEMIS_ENABLE_HSM ON)
    set(THEMIS_ENABLE_AUTO_FAILOVER ON)
    set(THEMIS_ENABLE_COMPLIANCE_AUDIT ON)
    set(THEMIS_ENABLE_MASSIVE_CLUSTERING ON)
    set(THEMIS_ENABLE_ADVANCED_GPU ON)
    
    message(STATUS "Building ThemisDB HYPERSCALER Edition")
    target_compile_definitions(themis_core PUBLIC THEMIS_EDITION_HYPERSCALER)
    add_subdirectory(src/enterprise)
    add_subdirectory(src/hyperscaler)
else()
    message(FATAL_ERROR "Unknown edition: ${THEMIS_EDITION}")
endif()
```

## 🔄 Release Pipeline (Automated)

### GitHub Actions Workflow

```yaml
# .github/workflows/v1.3.5-multi-edition-release.yml

name: v1.3.5 Multi-Edition Release Build

on:
  push:
    tags:
      - 'v1.3.5*'

jobs:
  # 1. Community Edition (Public Release)
  build-community:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      - name: Build Community (All Platforms)
        run: |
          ./.github/workflows/04-release_build-binary-linux.yml -Version "1.3.5"
      - name: Upload to GitHub Releases (Public)
        run: gh release create v1.3.5-community ...
      - name: Push to Docker Hub (Public)
        run: |
          docker login -u ${{ secrets.DOCKERHUB_USERNAME }} ...
          docker push themisdb:1.3.5-community
          docker push themisdb:latest

  # 2. Enterprise Edition (Private Release)
  build-enterprise:
    runs-on: ubuntu-latest
    if: secrets.ENTERPRISE_BUILD_KEY != ''
    steps:
      - uses: actions/checkout@v3
      - name: Build Enterprise (All Platforms)
        run: |
          ./.github/workflows/04-release_publish-enterprise.yml -Version "1.3.5"
      - name: Generate License Key Validator
        run: ./scripts/generate-license-validator.ps1
      - name: Upload to Private Release Portal
        run: ./scripts/upload-enterprise-packages.ps1
      - name: Push to Private Docker Registry
        run: |
          docker login -u ${{ secrets.PRIVATE_REGISTRY_USER }} ...
          docker push themisdb:1.3.5-enterprise

  # 3. Hyperscaler Edition (OEM Direct)
  build-hyperscaler:
    runs-on: ubuntu-latest
    if: secrets.HYPERSCALER_BUILD_KEY != ''
    steps:
      - uses: actions/checkout@v3
      - name: Build Hyperscaler (All Platforms)
        run: |
          ./.github/workflows/04-release_publish-hyperscaler.yml -Version "1.3.5"
      - name: Generate OEM Documentation
        run: ./scripts/generate-oem-docs.ps1
      - name: Push to Customer-Specific Registries
        run: ./scripts/deploy-oem-packages.ps1
```
