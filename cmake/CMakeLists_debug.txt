cmake_minimum_required(VERSION 3.20)
project(test_rocksdb_debug)

set(CMAKE_CXX_STANDARD 20)

set(_themis_debug_vcpkg_include_dir "")
set(_themis_debug_vcpkg_lib_dir "")
if(DEFINED VCPKG_INSTALLED_DIR AND DEFINED VCPKG_TARGET_TRIPLET)
    set(_themis_debug_vcpkg_include_dir "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/include")
    set(_themis_debug_vcpkg_lib_dir "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/lib")
endif()

# Find required packages
find_package(RocksDB CONFIG REQUIRED)

# Include directories
include_directories(
    ${CMAKE_SOURCE_DIR}/include
    ${_themis_debug_vcpkg_include_dir}
)

# Link directories
link_directories(
    ${CMAKE_SOURCE_DIR}/build-msvc/Release
    ${_themis_debug_vcpkg_lib_dir}
)

# Create executable
add_executable(test_debug test_debug.cpp)

# Link libraries
target_link_libraries(test_debug 
    themis_core
    RocksDB::rocksdb
)
