cmake_minimum_required(VERSION 3.20)
project(ThemisDB_RocksDB_Tests CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wall -Wextra")

# Find required dependencies
find_package(RocksDB REQUIRED)
find_package(GTest REQUIRED)
find_package(Threads REQUIRED)
find_package(fmt REQUIRED)
find_package(spdlog REQUIRED)

# Include directories
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/include)

# Add the RocksDB wrapper source along with necessary utilities
add_library(rocksdb_wrapper 
    src/storage/rocksdb_wrapper.cpp
    src/utils/logger.cpp
    src/utils/error_registry.cpp
    src/utils/build_info.cpp
)
target_link_libraries(rocksdb_wrapper 
    PRIVATE
        RocksDB::rocksdb
        Threads::Threads
        fmt::fmt
        spdlog::spdlog
)
target_include_directories(rocksdb_wrapper PUBLIC include)

# Create test executable
add_executable(test_lib_rocksdb_integration tests/test_lib_rocksdb_integration.cpp)
target_link_libraries(test_lib_rocksdb_integration 
    PRIVATE
        rocksdb_wrapper
        GTest::GTest
        GTest::Main
        Threads::Threads
        fmt::fmt
        spdlog::spdlog
)
target_include_directories(test_lib_rocksdb_integration PRIVATE include)

# Enable testing
enable_testing()
add_test(NAME RocksDBIntegration COMMAND test_lib_rocksdb_integration)
